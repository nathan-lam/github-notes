$$
\begin{aligned}
A &:= \text{Set called A}\\
a &:= \text{Element called a}\\
a\in A &:= \text{a is an element of A}\\
b\notin A &:= \text{b is an element NOT in A}\\
A \subset B &:= \text{Set A is a subset of set B}\\
A=B &:= \text{Set A and set B have the same elements}\\

\end{aligned}
$$
